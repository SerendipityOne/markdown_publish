    ROS2常用开发工具

    让ROS2成为机器人开发利器的，不仅仅是前面介绍过的通信机制，还有各种强大易用的工具。前面我们用到过的rqt就是ROS2常用的可视化工具之一，本章将着重介绍ROS2机器人开发过程中的常用工具。

    在机器人的开发过程中，坐标变换非常重要且不好处理，ROS2就基于话题通信设计了一套库和工具，用于管理机器人坐标变换。

    从代码清单5-8的结果可以看出，tf2msgs/msg/TFMessage主要由geometry_msgs/TransformStamped的数组构成。

    每个TransformStamped包含以下字段：

    1. Header部分
      • header: 标准消息头
        • stamp: 时间戳
          • sec: 秒部分（int32）
          • nanosec: 纳秒部分（uint32）
        • frame_id: 父坐标系名称（string）
    2. 坐标系信息
      • child_frame_id: 子坐标系名称（string）
    3. Transform变换信息
      • transform: 具体的变换数据
        • translation: 平移部分（Vector3）
          • x: x方向平移量（float64）
          • y: y方向平移量（float64）
          • z: z方向平移量（float64）
        • rotation: 旋转部分（Quaternion）
          • x: 四元数x分量（float64，默认0）
          • y: 四元数y分量（float64，默认0）
          • z: 四元数z分量（float64，默认0）
          • w: 四元数w分量（float64，默认1）

    Python中的手眼坐标变换

    通过Python发布静态TF

    ​ 图中机械臂底座和相机一般是固定起来的，它们之间的坐标关系固定不变，可以采用静态的TF进行发布。接着我们用Python编写节点来进行静态的TF发布。

    ​ 因为TF发布时，旋转表示要使用四元数，所以要使用代码将欧拉角转换成四元数形式， ROS2中tf_transformations库提供这一功能。

    sudo apt install ros-$ROS_DISTRO-tf-transformations

    在chapt5/chapt5_ws/src下运行

    ros2 pkg create demo_python_tf --build-type ament_python --dependencies rclpy geomerty_msgs tf_ros tf_transformations --license Apache-2.0

    上面的命令创建了demo_python_tf功能包。依赖tf_ros用于发布广播，geometry_msgs是发布坐标变换所需要的消息接口，tf_transformations用于提供姿态转换相关函数。创建完成后，在chapts5_ws/src/demo_python_tf/demo_python_tf下新建static_tf_broadcaster.py

    import rclpy
from rclpy.node import Node
from tf2_ros import StaticTransformBroadcaster
from geometry_msgs.msg import TransformStamped
from tf_transformations import quaternion_from_euler
import math


class StaticTfBroadcaster(Node):
    """
    静态坐标变换广播器节点类

    该类继承自ROS2的Node类，用于发布静态坐标变换信息。
    它创建一个从base_link到camera_link的静态坐标变换。
    """

    def __init__(self):
        """
        初始化静态坐标变换广播器节点

        创建StaticTransformBroadcaster实例并调用发布静态坐标变换的方法。
        """
        super().__init__("static_tf_broadcaster")
        self.static_broadcaster = StaticTransformBroadcaster(self)
        self.publish_static_tf()

    def publish_static_tf(self):
        """
        发布静态坐标变换

        创建并配置一个TransformStamped消息，设置从base_link到camera_link的变换关系，
        包括平移和旋转信息，并通过静态坐标变换广播器发布。
        """
        transform = TransformStamped()
        transform.header.stamp = self.get_clock().now().to_msg()
        transform.header.frame_id = "base_link"
        transform.child_frame_id = "camera_link"
        transform.transform.translation.x = 0.5
        transform.transform.translation.y = 0.3
        transform.transform.translation.z = 0.6
        # 欧拉角转四元组
        q = quaternion_from_euler(math.radians(180), 0.0, 0.0)
        transform.transform.rotation.x = q[0]
        transform.transform.rotation.y = q[1]
        transform.transform.rotation.z = q[2]
        transform.transform.rotation.w = q[3]

        # 发布静态坐标变换
        self.static_broadcaster.sendTransform(transform)
        self.get_logger().info(f"发布静态tf：{transform}")


def main(args=None):
    """
    主函数

    初始化ROS2客户端库，创建静态坐标变换广播器节点，
    运行节点并处理程序退出时的清理工作。

    Args:
        args: 命令行参数，默认为None
    """
    rclpy.init(args=args)
    static_tf_broadcaster = StaticTfBroadcaster()
    rclpy.spin(static_tf_broadcaster)
    static_tf_broadcaster.destroy_node()
    rclpy.shutdown()

    通过Python发布动态TF

    src/demo_python_tf/demo_python_tf/dynamic_tf_broadcaster.py

    import rclpy
from rclpy.node import Node
from tf2_ros import TransformBroadcaster
from geometry_msgs.msg import TransformStamped
from tf_transformations import quaternion_from_euler
import math


class DynamicTfBroadcaster(Node):
    """
    动态坐标变换广播器类

    该类继承自ROS2的Node类，用于定期发布坐标变换信息。
    它创建一个定时器，以固定频率发布从camera_link到bottle_link的坐标变换。
    """

    def __init__(self):
        """
        初始化动态坐标变换广播器节点

        创建TransformBroadcaster实例用于发布坐标变换，
        并设置定时器以100Hz频率调用publish_tf方法。
        """
        super().__init__("dynamic_tf_broadcaster")
        self.broadcaster = TransformBroadcaster(self)
        # 动态TF持续发送，发送频率100Hz
        self.timer = self.create_timer(0.01, self.publish_tf)

    def publish_tf(self):
        """
        发布坐标变换消息

        创建并填充一个TransformStamped消息，设置从camera_link到bottle_link的变换，
        包括平移和旋转信息，并通过TransformBroadcaster进行广播。
        """
        transform = TransformStamped()
        transform.header.stamp = self.get_clock().now().to_msg()
        transform.header.frame_id = "camera_link"
        transform.child_frame_id = "bottle_link"
        transform.transform.translation.x = 0.2
        transform.transform.translation.y = 0.0
        transform.transform.translation.z = 0.5
        # 欧拉角转四元组
        q = quaternion_from_euler(0, 0.0, 0.0)
        transform.transform.rotation.x = q[0]
        transform.transform.rotation.y = q[1]
        transform.transform.rotation.z = q[2]
        transform.transform.rotation.w = q[3]

        # 发布静态坐标变换
        self.broadcaster.sendTransform(transform)
        self.get_logger().info(f"发布动态tf：{transform}")


def main(args=None):
    """
    主函数

    初始化ROS2客户端库，创建静态坐标变换广播器节点，
    运行节点并处理程序退出时的清理工作。

    Args:
        args: 命令行参数，默认为None
    """
    rclpy.init(args=args)
    dynamic_tf_broadcaster = DynamicTfBroadcaster()
    rclpy.spin(dynamic_tf_broadcaster)
    dynamic_tf_broadcaster.destroy_node()
    rclpy.shutdown()

    通过Python查询TF关系

    src/demo_python_tf/demo_python_tf/tf_listener.py

    import rclpy
from rclpy.node import Node
from tf2_ros import TransformListener, Buffer
from geometry_msgs.msg import TransformStamped
from tf_transformations import euler_from_quaternion
import math


class TFListener(Node):
    """
    TF监听器类，用于监听和获取坐标变换信息

    该类继承自ROS2的Node类，创建一个TF监听器节点，
    定期查询两个坐标系之间的变换关系。
    """

    def __init__(self):
        """
        初始化TF监听器节点

        创建Buffer缓冲区和TransformListener监听器，
        并设置定时器定期执行坐标变换查询。
        """
        super().__init__("tf_listener")
        self.buffer = Buffer()
        self.listener = TransformListener(self)
        self.timer = self.create_timer(0.01, self.get_tf)

    def get_tf(self):
        """
        获取坐标变换函数

        查询从"base_link"到"bottle_link"的坐标变换，
        将四元数形式的旋转转换为欧拉角并输出结果。
        如果查询失败则输出警告信息。
        """
        try:
            # 查询坐标变换
            transform = self.buffer.lookup_transform(
                "base_link",
                "bottle_link",
                rclpy.time.Time(seconds=0.0),
                rclpy.time.Duration(seconds=1.0),
            ).transform
            # 将四元数转换为欧拉角
            rotation_euler = euler_from_quaternion(
                [
                    transform.rotation.x,
                    transform.rotation.y,
                    transform.rotation.z,
                    transform.rotation.w,
                ]
            )
            # 输出变换信息
            self.get_logger().info(
                f"平移：{transform.translation}，旋转四元组：{transform.rotation}，旋转欧拉角：{rotation_euler}"
            )
        except Exception as e:
            # 处理异常情况
            self.get_logger().warn(f"无法获取坐标变换，原因：{str(e)}")


def main(args=None):
    rclpy.init(args=args)
    tf_listener = TFListener()
    rclpy.spin(tf_listener)
    tf_listener.destroy_node()
    rclpy.shutdown()

    通过C++发布静态TF

    src/demo_cpp_tf/src/map_to_target_tf.cpp

    #include "geometry_msgs/msg/transform_stamped.hpp"  // 提供消息接口类
#include "rclcpp/rclcpp.hpp"
#include "tf2/LinearMath/Quaternion.h"               // 提供 tf2::Quaternion 类
#include "tf2_geometry_msgs/tf2_geometry_msgs.hpp"   // 提供消息类型转换函数
#include "tf2_ros/static_transform_broadcaster.hpp"  // 提供静态坐标广播器
using StaticTransformBroadcaster = tf2_ros::StaticTransformBroadcaster;

class MapToTargetStaticTF : public rclcpp::Node {
  private:
  std::shared_ptr<StaticTransformBroadcaster> broadcaster_;  // 创建静态坐标变换发布者

  void publishStaticTF() {
    geometry_msgs::msg::TransformStamped transform;
    transform.header.stamp = this->get_clock()->now();
    transform.header.frame_id = "map";
    transform.child_frame_id = "target_point";
    transform.transform.translation.x = 5.0;
    transform.transform.translation.y = 3.0;
    transform.transform.translation.z = 0.0;
    tf2::Quaternion q;
    q.setRPY(0.0, 0.0, 60 * M_PI / 180.0);  // 旋转角度为60度，欧拉角转四元组
    transform.transform.rotation = tf2::toMsg(q);
    broadcaster_->sendTransform(transform);
    // 打印坐标变换信息
    RCLCPP_INFO(this->get_logger(),
                "Static TF: map -> target_point (x: %f, y: %f, z: %f)",
                transform.transform.translation.x, transform.transform.translation.y, transform.transform.translation.z);
  }

  public:
  MapToTargetStaticTF() : Node("map_to_target_tf") {
    broadcaster_ = std::make_shared<StaticTransformBroadcaster>(this);
    this->publishStaticTF();
  }
};

int main(int argc, char* argv[]) {
  // 初始化ROS2
  rclcpp::init(argc, argv);

  // 创建静态坐标变换广播器节点并运行
  auto node = std::make_shared<MapToTargetStaticTF>();
  rclcpp::spin(node);

  // 清理资源
  rclcpp::shutdown();
  return 0;
}

    通过C++发布动态TF

    发布动态TF和发布静态TF相比，最大的不同在于动态T℉需要不断向外发布。

    src/demo_cpp_tf/src/map_to_base_tf.cpp

    #include "geometry_msgs/msg/transform_stamped.hpp"
#include "rclcpp/rclcpp.hpp"
#include "tf2/LinearMath/Quaternion.h"
#include "tf2_geometry_msgs/tf2_geometry_msgs.hpp"
#include "tf2_ros/transform_broadcaster.hpp"
using TransformBroadcaster = tf2_ros::TransformBroadcaster;
using namespace std::chrono_literals;  // 使用时间单位的字面量，可以在代码中使用s和ms表示时间

class MapToBaseDynamicTF : public rclcpp::Node {
  private:
  std::shared_ptr<TransformBroadcaster> broadcaster_;  // 创建坐标变换发布者
  rclcpp::TimerBase::SharedPtr timer_;

  public:
  MapToBaseDynamicTF() : Node("map_to_base_tf") {
    broadcaster_ = std::make_shared<TransformBroadcaster>(this);
    timer_ = this->create_wall_timer(10ms, [&]() -> void {
      geometry_msgs::msg::TransformStamped transform;

      transform.header.stamp = this->get_clock()->now();
      transform.header.frame_id = "map";
      transform.child_frame_id = "base_link";

      transform.transform.translation.x = 2.0;
      transform.transform.translation.y = 3.0;
      transform.transform.translation.z = 0.0;

      tf2::Quaternion q;
      q.setRPY(0.0, 0.0, 30 * M_PI / 180.0);
      transform.transform.rotation = tf2::toMsg(q);
      broadcaster_->sendTransform(transform);

      RCLCPP_INFO(this->get_logger(), "发布动态tf：%f %f %f",
                  transform.transform.translation.x,
                  transform.transform.translation.y,
                  transform.transform.translation.z);
    });
  }
};

int main(int argc, char* argv[]) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<MapToBaseDynamicTF>();
  rclcpp::spin(node);
  rclcpp::shutdown();
  return 0;
}

    通过C++查询TF关系

    src/demo_cpp_tf/src/target_base_listener.cpp

    #include "geometry_msgs/msg/transform_stamped.hpp"
#include "rclcpp/rclcpp.hpp"
#include "tf2/LinearMath/Quaternion.h"
#include "tf2/utils.hpp"  // 提供tf2::getEulerYPR()
#include "tf2_geometry_msgs/tf2_geometry_msgs.hpp"
#include "tf2_ros/buffer.hpp"
#include "tf2_ros/transform_listener.hpp"
using TransformListener = tf2_ros::TransformListener;
using namespace std::chrono_literals;

class MapToBaseTFListener : public rclcpp::Node {
  private:
  std::shared_ptr<TransformListener> listener_;  // 创建坐标变换发布者
  std::shared_ptr<tf2_ros::Buffer> buffer_;
  rclcpp::TimerBase::SharedPtr timer_;

  public:
  MapToBaseTFListener() : Node("map_to_base_tf_listener") {
    buffer_ = std::make_shared<tf2_ros::Buffer>(this->get_clock());
    listener_ = std::make_shared<TransformListener>(*buffer_);
    timer_ = this->create_wall_timer(5s, [&]() -> void {
      try {
        auto transform = this->buffer_->lookupTransform(
          "target_point", "base_link",
          this->get_clock()->now(),
          rclcpp::Duration(1s))
          .transform;
        // 转换结果
        auto& translation = transform.translation;
        auto& rotation = transform.rotation;
        double roll, pitch, yaw;
        tf2::getEulerYPR(rotation, yaw, pitch, roll);  // 四元数转欧拉角

        // 输出变换信息
        RCLCPP_INFO(get_logger(),
                    "平移：[x: %f, y: %f, z: %f], "
                    "旋转四元组：[x: %f, y: %f, z: %f, w: %f], "
                    "旋转欧拉角：[roll: %f, pitch: %f, yaw: %f]",
                    translation.x,
                    translation.y,
                    translation.z,
                    rotation.x,
                    rotation.y,
                    rotation.z,
                    rotation.w,
                    roll,
                    pitch,
                    yaw);

      } catch (const std::exception& e) {
        RCLCPP_WARN(get_logger(), "无法获取坐标变换，原因：%s", e.what());
      }
    });
  }
};

int main(int argc, char* argv[]) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<MapToBaseTFListener>();
  rclcpp::spin(node);
  rclcpp::shutdown();
  return 0;
}
