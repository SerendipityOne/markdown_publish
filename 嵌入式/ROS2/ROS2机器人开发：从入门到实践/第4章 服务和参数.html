    每一行代表一个服务，每一行的前面是服务的名称，[]则是服务的接口类型。所以服务和话题一样，都是有名字和接口的；但服务的接口与话题不同，分为请求接口和响应接口两部分。以生成海龟服务/spawn的接口turtlesim/srv/Spawn为例

    $ ros2 interface show turtlesim/srv/Spawn
float32 x
float32 y
float32 theta
string name # Optional.  A unique name will be created and returned if this is empty
---
string name

    在返回结果中，由“---”隔开的上半部分为请求接口的定义，下半部分为响应接口的定义。/spawn服务的作用是产生一只新的海龟，请求接口中x、y和theta表示新的海龟所在位置和朝向，name是一个可选参数，表示新海龟的名字。响应接口定义中name表示生成的新海龟的名字。

    $ ros2 service call /spawn turtlesim/srv/Spawn "{x: 1, y: 1}"
requester: making request: turtlesim.srv.Spawn_Request(x=1.0, y=1.0, theta=0.0, name='')

response:
turtlesim.srv.Spawn_Response(name='turtle2')

    service call命令用于调用指定的服务，其第一个参数是服务的名字，第二个参数是服务的接口类型，第三个参数是Request数据。上面的命令用于在x=1、y=1的位置产生一只新的海龟，结果返回新海龟的名字为turtle2。

    用Python服务通信实现人脸检测

    自定义服务接口

    服务端要进行人脸检测，在客户端的请求中就需要带图片信息，而服务端完成人脸检测后，应该将人脸在图片中的位置信息返回给客户端，因为同一张图片中有多张人脸，所以需要用数组来表示人脸。在ROS2已有的服务接口中，没有能满足这一要求的，所以需要自定义服务接口。

    chapt4/src

    $ ros2 pkg create interfaces --build-type ament_cmake --dependencies rosidl_default_generators sensor_msgs --license Apache-2.0

    src/interfaces/srv/FaceDetector.srv

    sensor_msgs/Image image  # 原始图像
---
int16 number        # 人脸数
float32 use_time    # 识别耗时
int32[] top         # 人脸在图像中的位置
int32[] right
int32[] bottom
int32[] left

    在上面的接口定义中，“---”上为Request部分，定义了一个sensor_msgs/Image类型的 image用于表示图像。“---”下为Response部分，number表示人脸数，use_time表示识别耗时，后面的四个数组表示每张人脸在图像中的位置信息，单位是像素。CMakeLists.txt代码

    find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(sensor_msgs REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
    "srv/FaceDetector.srv"
    DEPENDENCIES sensor_msgs
)

    人脸检测服务实现

    src/demo_python_service/demo_python_service/face_detect_node.py

    import rclpy
from rclpy.node import Node
from interfaces.srv import FaceDetector
from ament_index_python.packages import get_package_share_directory
import face_recognition
from cv_bridge import CvBridge
import cv2
import time
import os


class FaceDetectorNode(Node):
    def __init__(self):
        super().__init__("face_detect_node")
        self.bridge = CvBridge()
        # 创建服务
        self.service = self.create_service(
            FaceDetector, "face_detect", self.detect_face_callback
        )
        self.defalut_image_path = os.path.join(
            get_package_share_directory("demo_python_service"), "resource/image.png"
        )  # 默认图片路径
        self.number_of_times_to_upsample = 1
        self.model = "hog"
        self.get_logger().info(f"[FaceDetectorNode]初始化成功，开始启动")

    def detect_face_callback(self, request, response):
        if request.image.data:
            cv_image = self.bridge.imgmsg_to_cv2(request.image)
        else:  # 如果传入图片为空，则设置默认图片识别
            cv_image = cv2.imread(self.defalut_image_path)
            self.get_logger().info(f"传入图片能为空，设置默认图片识别")

        self.get_logger().info(f"开始识别")
        start_time = time.time()
        face_locations = face_recognition.face_locations(
            cv_image,
            number_of_times_to_upsample=self.number_of_times_to_upsample,
            model=self.model,
        )
        end_time = time.time()
        self.get_logger().info(f"识别完成，耗时：{end_time - start_time}秒")


        """ 
        int16 number            # 人脸数
        float32 use_time        # 识别耗时
        int32[] top             # 人脸在图像中的位置
        int32[] right
        int32[] bottom
        int32[] left
        """
        # 给response赋值
        response.number = len(face_locations)
        response.use_time = end_time - start_time
        for top, right, bottom, left in face_locations:
            response.top.append(top)
            response.right.append(right)
            response.bottom.append(bottom)
            response.left.append(left)

        return response  # 需返回response给客户端


def main():
    rclpy.init()
    node = FaceDetectorNode()
    rclpy.spin(node)
    rclpy.shutdown()

    src/demo_python_service/demo_python_service/face_detect_client_node.py

    import rclpy
from rclpy.node import Node
from interfaces.srv import FaceDetector
from ament_index_python.packages import get_package_share_directory
from cv_bridge import CvBridge
import cv2
import os


class FaceDetectClientNode(Node):
    def __init__(self):
        super().__init__("face_detect_client_node")
        # 创建客户端
        self.client = self.create_client(FaceDetector, "face_detect")
        self.bridge = CvBridge()
        self.defalut_image_path = os.path.join(
            get_package_share_directory("demo_python_service"), "resource/test1.jpg"
        )  # 默认图片路径
        self.image = cv2.imread(self.defalut_image_path)
        self.get_logger().info(f"[FaceDetectClientNode]初始化成功，开始启动")

    def send_request(self):
        # 1.判断服务端是否在线
        while not self.client.wait_for_service(timeout_sec=1.0):
            self.get_logger().info("服务端未在线，请稍后...")
        # 2.构造Request
        request = FaceDetector.Request()
        request.image = self.bridge.cv2_to_imgmsg(self.image)
        # 4.发送请求并等待处理完成
        future = self.client.call_async(
            request
        )  # 现在的future并没有包含响应结果，需要等待服务端处理完成才会把结果放到future中
        self.get_logger().info("发送请求成功，等待响应...")

        # while not future.done():
        #     time.sleep(1.0) # 休眠当前线程，等待服务处理完成==造成当前线程无法再接受来自服务端的返回，导致永远没有办法完成-->future.done无法返回True

        def result_callback(result_future):
            """
            处理服务端返回结果的回调函数

            Args:
                result_future: 包含服务端响应结果的Future对象
            """
            rclpy.spin_until_future_complete(self, future)  # 等待服务调用完成
            response = result_future.result()
            self.get_logger().info(
                f"接收到响应，共检测到有{response.number}张人脸，耗时{response.use_time}s"
            )
            self.show_response(response)

        future.add_done_callback(result_callback)

    def show_response(self, response):
        for i in range(response.number):
            top = response.top[i]
            right = response.right[i]
            bottom = response.bottom[i]
            left = response.left[i]
            cv2.rectangle(self.image, (left, top), (right, bottom), (0, 255, 0), 4)
        # 显示结果图像
        cv2.imshow("Face Detection Result", self.image)
        # 这里会阻塞，但我们只发送一次请求，所以没事，如果是多次请求则不要加
        cv2.waitKey(0)


def main():
    rclpy.init()
    node = FaceDetectClientNode()
    node.send_request()
    rclpy.spin(node)
    rclpy.shutdown()

    服务端代码实现

    src/demo_cpp_service/src/turtle_control.cpp

    #include "geometry_msgs/msg/twist.hpp"
#include "interfaces/srv/patrol.hpp"
#include "rclcpp/rclcpp.hpp"
#include "turtlesim/msg/pose.hpp"
// 使用时间单位的字面量，可以在代码中使用s和ms表示时间
using namespace std::chrono_literals;
using Patrol = interfaces::srv::Patrol;

class TurtleControlNode : public rclcpp::Node {
   private:
    rclcpp::Service<Patrol>::SharedPtr service;                          // 服务提供者智能指针
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr publisher;   // 发布者智能指针
    rclcpp::Subscription<turtlesim::msg::Pose>::SharedPtr subscription;  // 订阅者智能指针
    // 目标坐标
    double target_x = 1.0, target_y = 1.0;
    // 控制参数，比例系数
    double k = 2.0;
    double max_linear_speed = 2.0;

   public:
    explicit TurtleControlNode(const std::string node_name) : Node(node_name) {
        // 调用继承而来的父类函数创建订阅者
        service = this->create_service<Patrol>(
            "/patrol",
            [&](const Patrol::Request::SharedPtr request, Patrol::Response::SharedPtr response) -> void {
                // 在范围内则设置目标点
                if ((request->target_x > 0 && request->target_x < 12) &&
                    (request->target_y > 0 && request->target_y < 12)) {
                    target_x = request->target_x;
                    target_y = request->target_y;
                    response->result = Patrol::Response::SUCCESS;
                    RCLCPP_INFO(this->get_logger(), "目标点设置成功，target_x: %f, target_y: %f", target_x, target_y);
                } else {
                    response->result = Patrol::Response::FAIL;
                    RCLCPP_INFO(this->get_logger(), "超出范围目标点设置失败");
                }
            });
        publisher = this->create_publisher<geometry_msgs::msg::Twist>("/turtle1/cmd_vel", 10);
        subscription = this->create_subscription<turtlesim::msg::Pose>(
            "/turtle1/pose", 10,
            std::bind(&TurtleControlNode::on_pose_recieved,
                      this, std::placeholders::_1));
        RCLCPP_INFO(this->get_logger(), "[turtle_control]初始化成功");
    }

   private:
    void on_pose_recieved(const turtlesim::msg::Pose::SharedPtr pose) {  // 参数：收到数据的共享指针
        // 1.获取当前位置
        auto cur_x = pose->x, cur_y = pose->y;
        RCLCPP_INFO(get_logger(), "cur_x: %f, cur_y: %f", cur_x, cur_y);

        // 2.计算当前海龟位置跟目标位置之间的距离差和角度差
        // 计算当前点(cur_x, cur_y)到目标点(target_x, target_y)的直线距离
        auto dist = std::sqrt(
            (target_x - cur_x) * (target_x - cur_x) +
            (target_y - cur_y) * (target_y - cur_y));
        // 使用atan2反正切函数计算从当前位置到目标位置的方位角，然后减去当前朝向角
        auto angle = std::atan2(target_y - cur_y, target_x - cur_x) - pose->theta;

        // 3.控制
        auto msg = geometry_msgs::msg::Twist();
        if (dist > 0.1) {
            if (fabs(angle) > 0.2) {
                msg.angular.z = fabs(angle);  // 当前角度差大于0.2时，设置角速度
            } else {
                msg.linear.x = k * dist;  // 当前角度差小于0.2时，设置线速度
            }
        }

        // 限制最大线速度
        msg.linear.x = std::min(max_linear_speed, msg.linear.x);

        publisher->publish(msg);
    }
};

int main(int argc, char const* argv[]) {
    rclcpp::init(argc, argv);
    auto node = std::make_shared<TurtleControlNode>("turtle_control");
    rclcpp::spin(node);
    rclcpp::shutdown();
    return 0;
}

 * @brief 巡逻客户端节点类

class PatrolClientNode : public rclcpp::Node {
  private:
  rclcpp::Client<Patrol>::SharedPtr client;
  rclcpp::TimerBase::SharedPtr timer;

  public:
  /**
     * @brief 构造函数，初始化PatrolClientNode节点
     * 
     * 在构造函数中完成以下任务：
     * 1. 初始化随机数种子
     * 2. 创建巡逻服务的客户端
     * 3. 创建定时器，每10秒触发一次服务请求
     * 4. 在定时器回调中处理服务连接和服务请求发送
     */
  explicit PatrolClientNode() : Node("PatrolClientNode") {
    srand(time(NULL));  // 设置随机数种子
    this->client = this->create_client<Patrol>("/patrol");

    // 创建一个定时器，每隔10秒执行一次lambda表达式
    this->timer = this->create_wall_timer(10s, [&]() -> void {
      // 检测服务是否上线
      while (!this->client->wait_for_service(1s)) {
        if (!rclcpp::ok()) {
          RCLCPP_ERROR(this->get_logger(), "在等待服务的过程中被中断了。正在退出。");
          return;
        }
        RCLCPP_INFO(this->get_logger(), "等待服务上线中...");
      }

      // 构造请求对象
      auto request = std::make_shared<Patrol::Request>();
      request->target_x = rand() % 15; // 生成0到14之间的随机数，后续发送请求到服务端赋值给服务端的target_x
      request->target_y = rand() % 15;
      RCLCPP_INFO(this->get_logger(), "请求巡逻：(%f, %f)", request->target_x, request->target_y);

      // 发送异步请求
      this->client->async_send_request(request, [&](rclcpp::Client<Patrol>::SharedFuture result_future) -> void {
        auto response = result_future.get();
        if (response->result == Patrol::Response::SUCCESS) {
          RCLCPP_INFO(this->get_logger(), "目标点设置成功");
        } else if (response->result == Patrol::Response::FAIL) {
          RCLCPP_INFO(this->get_logger(), "目标点设置失败");
        }
      });
    });
  }
};

int main(int argc, char const* argv[]) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<PatrolClientNode>();
  rclcpp::spin(node);
  rclcpp::shutdown();
  return 0;

using namespace std::chrono_literals;
using Patrol = interfaces::srv::Patrol;

/**
 * @brief 巡逻客户端节点类
 * 
 * 该类继承自rclcpp::Node，用于创建一个ROS2节点，
 * 定期向巡逻服务发送随机坐标请求。
 */
class PatrolClientNode : public rclcpp::Node {
  private:
  rclcpp::Client<Patrol>::SharedPtr client;
  rclcpp::TimerBase::SharedPtr timer;
  std::shared_ptr<SetParameters::Response> call_set_params(Parameter& param) {
    // 1. 创建客户端等待服务上线
    auto param_client = this->create_client<SetParameters>("/turtle_control/set_parameters");
    while (!param_client->wait_for_service(std::chrono::seconds(1))) {
      RCLCPP_INFO(this->get_logger(), "等待服务端上线");
    }
    // 2. 创建请求
    auto request = std::make_shared<SetParameters::Request>();
    request->parameters.push_back(param);
    // 3. 发送请求
    auto future = param_client->async_send_request(request);
    rclcpp::spin_until_future_complete(this->get_node_base_interface(), future);
    auto response = future.get();
    return response;
  };

  public:
  /**
     * @brief 构造函数，初始化PatrolClientNode节点
     * 
     * 在构造函数中完成以下任务：
     * 1. 初始化随机数种子
     * 2. 创建巡逻服务的客户端
     * 3. 创建定时器，每10秒触发一次服务请求
     * 4. 在定时器回调中处理服务连接和服务请求发送
     */
  explicit PatrolClientNode() : Node("PatrolClientNode") {
    srand(time(NULL));  // 设置随机数种子
    this->client = this->create_client<Patrol>("/patrol");

    // 创建一个定时器，每隔10秒执行一次lambda表达式
    this->timer = this->create_wall_timer(10s, [&]() -> void {
      // 检测服务是否上线
      while (!this->client->wait_for_service(1s)) {
        if (!rclcpp::ok()) {
          RCLCPP_ERROR(this->get_logger(), "在等待服务的过程中被中断了。正在退出。");
          return;
        }
        RCLCPP_INFO(this->get_logger(), "等待服务上线中...");
      }

      // 构造请求对象
      auto request = std::make_shared<Patrol::Request>();
      request->target_x = rand() % 15;  // 生成0到14之间的随机数，后续发送请求到服务端赋值给服务端的target_x
      request->target_y = rand() % 15;
      RCLCPP_INFO(this->get_logger(), "请求巡逻：(%f, %f)", request->target_x, request->target_y);

      // 发送异步请求
      this->client->async_send_request(request, [&](rclcpp::Client<Patrol>::SharedFuture result_future) -> void {
        auto response = result_future.get();
        if (response->result == Patrol::Response::SUCCESS) {
          RCLCPP_INFO(this->get_logger(), "目标点设置成功");
        } else if (response->result == Patrol::Response::FAIL) {
          RCLCPP_INFO(this->get_logger(), "目标点设置失败");
        }
      });
    });
  }

  void update_server_param_k(double k) {
    // 1.创建一个参数对象
    auto param = Parameter();
    param.name = "k";
    // 2.创建参数值对象并赋值
    auto param_val = ParameterValue();
    param_val.type = ParameterType::PARAMETER_DOUBLE;
    param_val.double_value = k;
    param.value = param_val;
    // 3.请求更新参数
    auto response = call_set_params(param);
    if (response == nullptr) {
      RCLCPP_INFO(this->get_logger(), "参数修改失败");
      return;

    } else {
      for (auto result : response->results) {
        if (result.successful) {
          RCLCPP_INFO(this->get_logger(), "k->%f", k);
        } else {
          RCLCPP_WARN(this->get_logger(), "参数k修改失败，原因：%s", result.reason.c_str());
        }
      }
    }
  }
};

int main(int argc, char const* argv[]) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<PatrolClientNode>();
  node->update_server_param_k(2);
  rclcpp::spin(node);
  rclcpp::shutdown();
  return 0;
}

    Launch进阶

    要掌握launch并灵活使用，你需要了解动作、条件和替换这三个launch组件的使用。下面我将为你提供示例代码，请你逐一编写，自行运行测试。首先来看动作，前面用的launch_ros.actions.Node就属于动作，动作除了是一个节点外，还可以是一句输出、一段终端命令，甚至是另外一个launch文件。launch中动作的更多使用方法（比如包含其他launch、执行进程、输出日志、组合和定时启动）

    src/demo_cpp_service/launch/action.launch.py

    import launch
import launch_ros
from ament_index_python.packages import get_package_share_directory


def generate_launch_description():
    # 启动其他action
    multisim_path = [
        get_package_share_directory("turtlesim"),
        "/launch",
        "/multisim.launch.py",
    ]
    action_multisim_launch = launch.actions.IncludeLaunchDescription(
        launch.launch_description_sources.PythonLaunchDescriptionSource(multisim_path)
    )

    # 打印数据
    action_log_info = launch.actions.LogInfo(msg=str(multisim_path))

    # 执行进程
    action_topic_list = launch.actions.ExecuteProcess(
        cmd=[
            "ros2",
            "topic",
            "list",
        ],
        output="screen",
    )

    # 组织启动
    action_group = launch.actions.GroupAction(
        actions=[
            # 定时器
            launch.actions.TimerAction(period=2.0, actions=[action_multisim_launch]),
            launch.actions.TimerAction(period=4.0, actions=[action_topic_list]),
        ]
    )

    return launch.LaunchDescription([action_log_info, action_group])
