    一、模块驱动

    1.1 OLED

    原理图：

    CubeMX设置PB9（SDA）和PB8（SCL）都设置为GPIO_Output：

    • GPIO mode: Output Open Drain（开漏输出）
      • GPIO Pull-up: Pull-up（使能内部上拉，或外部加4.7k上拉电阻）
      • Maximum output speed: Low或High（根据实际需求）

    ⚠️ 注意：软件I2C必须用开漏+上拉，否则无法正常通信。

    1.2 陀螺仪MPU6050

    此处需要用到MPU6050的DMP模式。DMP，即Digital Motion Processor（数字运动处理器），是MPU6050内部集成的一个硬件运算单元，能够独立完成复杂的运动数据融合、姿态解算（如四元数、欧拉角）、滤波、校准等任务，而无需外部MCU参与大量计算。

    从图中看出为使用软件模拟I2C，硬件I2C1为PB6/PB7，I2C2为PB10/PB11。

    • PB3（SDA）和PB4（SCL）都设置为：
      • GPIO mode: Output Open Drain（开漏输出）
      • GPIO Pull-up: Pull-up（使能内部上拉，或外部加4.7k上拉电阻）
      • Maximum output speed: Low或High（根据实际需求）

    中断引脚（PB5/INT），PB5设置为GPIO_EXTI5。在NVIC Settings选项卡中，勾选EXTI line[5:9] interrupts，使能中断。

    超详细陀螺仪MPU6050模块输出姿态角

    关于加速度与角速度的变化方向：假设Y轴正方向为机头，当机头上下时，改变的是Y轴的加速度以及X轴的角速度，而Y轴的角速度则是机头左右时改变。角速度是绕哪个轴旋转，哪个轴的角速度读数就会变化。

    使用DMP对姿态角进行滤波等算法优化得到，运算过程较为复杂，此处不做说明。由于获取姿态角需要经过计算得到，所以不适合在中断内读取数据。

    使用正点原子提供的DMP驱动，并自己添加MPU6050驱动，源代码在Github仓库中MPU6050

/**
 * @brief 初始化MPU6050传感器
 * @return 0表示初始化成功，1表示初始化失败
 */
uint8_t MPU_Init(void) {
    uint8_t res;  // 用于存储读取的设备ID

    MPU_IIC_Init();                           //初始化IIC总线
    MPU_Write_Byte(MPU_PWR_MGMT1_REG, 0X80);  //复位MPU6050
    delay_ms(100);                            //等待复位完成
    MPU_Write_Byte(MPU_PWR_MGMT1_REG, 0X00);  //唤醒MPU6050
    MPU_Set_Gyro_Fsr(3);                      //陀螺仪传感器,±2000dps
    MPU_Set_Accel_Fsr(0);                     //加速度传感器,±2g
    MPU_Set_Rate(100);                        //设置采样率100Hz
    MPU_Write_Byte(MPU_INT_EN_REG, 0X00);     //关闭所有中断
    MPU_Write_Byte(MPU_USER_CTRL_REG, 0X00);  //I2C主模式关闭
    MPU_Write_Byte(MPU_FIFO_EN_REG, 0X00);    //关闭FIFO
    MPU_Write_Byte(MPU_INTBP_CFG_REG, 0X80);  //INT引脚低电平有效
    res = MPU_Read_Byte(MPU_DEVICE_ID_REG);
    if (res == MPU_ADDR)  //器件ID正确
    {
        MPU_Write_Byte(MPU_PWR_MGMT1_REG, 0X01);  //设置CLKSEL,PLL X轴为参考
        MPU_Write_Byte(MPU_PWR_MGMT2_REG, 0X00);  //加速度与陀螺仪都工作
        MPU_Set_Rate(100);                        //设置采样率为100Hz
    } else
        return 1;
    return 0;
}

/**
 * @brief MPU_DMP初始化函数
 * @return uint8_t 初始化状态码，0表示成功，非0表示失败
 */
uint8_t MPU_DMP_Init(void) {
    uint8_t res;  // 用于存储初始化结果的状态变量

    // 初始化MPU传感器
    res = MPU_Init();
    if (res != 0) {
        return res;  // 如果MPU初始化失败，直接返回错误码
    }
    // 初始化MPU的DMP（数字运动处理器）功能
    res = mpu_dmp_init();
    if (res != 0) {
        return res;  // 如果DMP初始化失败，直接返回错误码
    }
    return 0;  // 所有初始化成功，返回0表示成功
}

/**
 * @brief 设置陀螺仪满量程范围
 * @param fsr 陀螺仪满量程范围设置值 fsr:0,±250dps;1,±500dps;2,±1000dps;3,±2000dps
 * @return uint8_t 操作状态，0表示成功，非0表示失败
 */
uint8_t MPU_Set_Gyro_Fsr(uint8_t fsr) {
    return MPU_Write_Byte(MPU_GYRO_CFG_REG, fsr << 3);  //设置陀螺仪满量程范围
}

/**
 * 设置MPU6050加速度传感器满量程范围
 * @param fsr 满量程范围参数(0-3对应±2g/±4g/±8g/±16g)
 * @return 0-设置成功, 其他-设置失败
 */
uint8_t MPU_Set_Accel_Fsr(uint8_t fsr) {
    return MPU_Write_Byte(MPU_ACCEL_CFG_REG, fsr << 3);  //设置加速度传感器满量程范围
}

/**
 * @brief 设置MPU6050的数字低通滤波器
 * @param lpf 数字低通滤波频率(Hz)
 * @return 0表示设置成功，其他值表示设置失败
 */
uint8_t MPU_Set_LPF(uint16_t lpf) {
    uint8_t data = 0;
    if (lpf >= 188)
        data = 1;
    else if (lpf >= 98)
        data = 2;
    else if (lpf >= 42)
        data = 3;
    else if (lpf >= 20)
        data = 4;
    else if (lpf >= 10)
        data = 5;
    else
        data = 6;
    return MPU_Write_Byte(MPU_CFG_REG, data);  //设置数字低通滤波器
}

/**
 * @brief 设置MPU6050的采样率(假定Fs=1KHz)
 * @param rate 采样率，范围为4~1000(Hz)
 * @return 0表示设置成功，其他值表示设置失败
 */
uint8_t MPU_Set_Rate(uint16_t rate) {
    uint8_t data;
    if (rate > 1000) rate = 1000;
    if (rate < 4) rate = 4;
    data = 1000 / rate - 1;
    data = MPU_Write_Byte(MPU_SAMPLE_RATE_REG, data);  //设置数字低通滤波器
    return MPU_Set_LPF(rate / 2);                      //自动设置LPF为采样率的一半
}

/**
 * @brief 得到温度值
 * @return 温度值(扩大了100倍)
 */
short MPU_Get_Temperature(void) {
    uint8_t buf[2];
    short raw;
    float temp;
    MPU_Read_Len(MPU_ADDR, MPU_TEMP_OUTH_REG, 2, buf);
    raw = ((uint16_t)buf[0] << 8) | buf[1];
    temp = 36.53 + ((double)raw) / 340;
    return temp * 100;
    ;
}

/**
 * @brief 得到陀螺仪值(原始值)
 * @param mpu 指向MPU结构体的指针，用于存储陀螺仪x,y,z轴的原始读数
 * @return 0表示读取成功，其他值为错误代码
 */
uint8_t MPU_Get_Gyroscope(MPU* mpu) {
    uint8_t buf[6], res;
    res = MPU_Read_Len(MPU_ADDR, MPU_GYRO_XOUTH_REG, 6, buf);
    if (res == 0) {
        mpu->gyrox = ((uint16_t)buf[0] << 8) | buf[1];
        mpu->gyroy = ((uint16_t)buf[2] << 8) | buf[3];
        mpu->gyroz = ((uint16_t)buf[4] << 8) | buf[5];
    }
    return res;
}

/**
 * @brief 得到加速度值(原始值)
 * @param mpu 指向MPU结构体的指针，用于存储加速度x,y,z轴的原始读数
 * @return 0表示读取成功，其他值为错误代码
 */
uint8_t MPU_Get_Accelerometer(MPU* mpu) {
    uint8_t buf[6], res;
    res = MPU_Read_Len(MPU_ADDR, MPU_ACCEL_XOUTH_REG, 6, buf);
    if (res == 0) {
        mpu->accelx = ((uint16_t)buf[0] << 8) | buf[1];
        mpu->accely = ((uint16_t)buf[2] << 8) | buf[3];
        mpu->accelz = ((uint16_t)buf[4] << 8) | buf[5];
    }
    return res;
    ;
}

/**
 * @brief IIC连续写
 * @param addr 器件地址
 * @param reg 寄存器地址
 * @param len 写入长度
 * @param buf 数据区指针
 * @return 0表示正常，其他值为错误代码
 */
uint8_t MPU_Write_Len(uint8_t addr, uint8_t reg, uint8_t len, uint8_t* buf) {
    uint8_t i;
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((addr << 1) | 0);  //发送器件地址+写命令
    if (MPU_IIC_Wait_Ack())              //等待应答
    {
        MPU_IIC_Stop();
        return 1;
    }
    MPU_IIC_Send_Byte(reg);  //写寄存器地址
    MPU_IIC_Wait_Ack();      //等待应答
    for (i = 0; i < len; i++) {
        MPU_IIC_Send_Byte(buf[i]);  //发送数据
        if (MPU_IIC_Wait_Ack())     //等待ACK
        {
            MPU_IIC_Stop();
            return 1;
        }
    }
    MPU_IIC_Stop();
    return 0;
}

/**
 * @brief IIC连续读
 * @param addr 器件地址
 * @param reg 要读取的寄存器地址
 * @param len 要读取的长度
 * @param buf 读取到的数据存储区
 * @return 0表示正常，其他值为错误代码
 */
uint8_t MPU_Read_Len(uint8_t addr, uint8_t reg, uint8_t len, uint8_t* buf) {
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((addr << 1) | 0);  //发送器件地址+写命令
    if (MPU_IIC_Wait_Ack())              //等待应答
    {
        MPU_IIC_Stop();
        return 1;
    }
    MPU_IIC_Send_Byte(reg);  //写寄存器地址
    MPU_IIC_Wait_Ack();      //等待应答
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((addr << 1) | 1);  //发送器件地址+读命令
    MPU_IIC_Wait_Ack();                  //等待应答
    while (len) {
        if (len == 1)
            *buf = MPU_IIC_Read_Byte(0);  //读数据,发送nACK
        else
            *buf = MPU_IIC_Read_Byte(1);  //读数据,发送ACK
        len--;
        buf++;
    }
    MPU_IIC_Stop();  //产生一个停止条件
    return 0;
}

/**
 * @brief IIC写一个字节
 * @param reg 寄存器地址
 * @param data 数据
 * @return 0表示正常，其他值为错误代码
 */
uint8_t MPU_Write_Byte(uint8_t reg, uint8_t data) {
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((MPU_ADDR << 1) | 0);  //发送器件地址+写命令
    if (MPU_IIC_Wait_Ack())                  //等待应答
    {
        MPU_IIC_Stop();
        return 1;
    }
    MPU_IIC_Send_Byte(reg);   //写寄存器地址
    MPU_IIC_Wait_Ack();       //等待应答
    MPU_IIC_Send_Byte(data);  //发送数据
    if (MPU_IIC_Wait_Ack())   //等待ACK
    {
        MPU_IIC_Stop();
        return 1;
    }
    MPU_IIC_Stop();
    return 0;
}

/**
 * @brief IIC读一个字节
 * @param reg 寄存器地址
 * @return 读到的数据
 */
uint8_t MPU_Read_Byte(uint8_t reg) {
    uint8_t res;
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((MPU_ADDR << 1) | 0);  //发送器件地址+写命令
    MPU_IIC_Wait_Ack();                      //等待应答
    MPU_IIC_Send_Byte(reg);                  //写寄存器地址
    MPU_IIC_Wait_Ack();                      //等待应答
    MPU_IIC_Start();
    MPU_IIC_Send_Byte((MPU_ADDR << 1) | 1);  //发送器件地址+读命令
    MPU_IIC_Wait_Ack();                      //等待应答
    res = MPU_IIC_Read_Byte(0);              //读取数据,发送nACK
    MPU_IIC_Stop();                          //产生一个停止条件
    return res;
}

/**
 * @brief  通过DMP获取姿态数据并填充到结构体中
 * @param  mpu: 指向用于存储数据的结构体
 * @retval 0: 成功
 *         1: FIFO读取失败
 *         2: 无四元数数据
 */
uint8_t MPU_DMP_ReadData(MPU* mpu) {
    float q0 = 1.0f, q1 = 0.0f, q2 = 0.0f, q3 = 0.0f;
    unsigned long sensor_timestamp;
    short gyro[3], accel[3], sensors;
    unsigned char more;
    long quat[4];
    if (dmp_read_fifo(gyro, accel, quat, &sensor_timestamp, &sensors, &more)) return 1;
    if (sensors & INV_WXYZ_QUAT) {
        q0 = quat[0] / q30;  //q30格式转换为浮点数
        q1 = quat[1] / q30;
        q2 = quat[2] / q30;
        q3 = quat[3] / q30;
        //计算得到俯仰角/横滚角/航向角
        mpu->pitch = asin(-2 * q1 * q3 + 2 * q0 * q2) * 57.3;                                     // pitch
        mpu->roll = atan2(2 * q2 * q3 + 2 * q0 * q1, -2 * q1 * q1 - 2 * q2 * q2 + 1) * 57.3;      // roll
        mpu->yaw = atan2(2 * (q1 * q2 + q0 * q3), q0 * q0 + q1 * q1 - q2 * q2 - q3 * q3) * 57.3;  //yaw

        // 获取原始值
        mpu->gyrox = gyro[0];
        mpu->gyroy = gyro[1];
        mpu->gyroz = gyro[2];

        mpu->accelx = accel[0];
        mpu->accely = accel[1];
        mpu->accelz = accel[2];
    } else
        return 2;
    return 0;
}

    1.4 电机编码器

    设置PA0/1为TIM2_CH，PB6/7为TIM4_CH设置Combined Channels：Encoder Mode并设置一个通道上升沿触发一个通道下降沿触发

    Input Filter（输入滤波器） 是用于对编码器输入信号进行硬件滤波的参数，主要用于消除信号抖动、噪声以及因机械振动或电气干扰引起的毛刺，从而提高测速的稳定性和准确性。

    参数取值范围

    • Input Filter 通常是一个 N 值，代表连续 N 次采样电平一致后才确认为有效边沿，常见取值范围：0~15（具体以芯片手册为准）。
    • 0：表示无滤波，直接采样。
    • 数值越大：滤波效果越强，但信号延迟也越大。

    推荐设置

    • 一般环境：建议 3~6。
    • 强干扰/高抖动：可设为 8~10。
    • 低速高精度：建议 2~4，避免滤波导致延迟过大。
    • 高速测速：可适当提高，如 6~8，但需测试延迟是否可接受。

    1.5 超声波模块HC-SR04

    HC-SR04超声波模块介绍

    正常情况下想要精准测距最佳方式是使用定时器的输入捕获功能。由于PA2复用TIM2_CH3但TIM2_CH1/2已经用作电机编码器Encoder，无法再用作输入捕获。退而求其次使用外部中断（EXTI） + 空闲定时器（TIM3） 的方式实现HC-SR04的ECHO脉冲宽度测量。

    1. 将PA3配置为 GPIO_Output

    2. PA2选择 GPIO_EXTI2

      • GPIO mode: External Interrupt Mode with Rising/Falling edge trigger detection

      • Pull-up/Pull-down: Pull down（防止悬空误触发） 常态低电平：HC-SR04的ECHO引脚在空闲时（未触发测距）默认输出低电平。 测距时高电平：触发TRIG后，ECHO会输出一个高电平脉冲，脉冲宽度等于超声波往返时间。 测距结束恢复低电平：脉冲结束后，ECHO回到低电平。

      • 开启中断

    3. 配置TIM3使用Internal Clock并设置PSC=72-1，ARR=65535，1us一次计数

    HR_SR04.c

    #include "HC_SR04.h"
#include "Delay.h"

extern TIM_HandleTypeDef htim3;

uint32_t echo_start = 0;  // 上升沿时刻的TIM3计数值
uint32_t echo_end = 0;    // 下降沿时刻的TIM3计数值
uint8_t echo_state = 0;   // 0:等待上升沿, 1:等待下降沿
float temp_distance = 0;  // 测量的距离值

/**
  * @brief  HC-SR04超声波传感器初始化函数
  * @param  无
  * @retval 无
  * @note   启动TIM3定时器，用于测量超声波回波脉冲宽度
  */
void HC_SR04_Init(void) {
  HAL_TIM_Base_Start(&htim3);
}

/**
  * @brief  获取HC-SR04超声波传感器测量的距离
  * @param  无
  * @retval float 测量的距离值，单位为厘米(cm)
  * @note   根据脉冲宽度计算距离，声波速度为340m/s，即0.034cm/us
  */
float HC_SR04_GetDistance(void) {
  // 发送触发信号
  HAL_GPIO_WritePin(SR04_TRIG_GPIO_Port, SR04_TRIG_Pin, GPIO_PIN_SET);
  Delay_us(10);  // 至少10us
  HAL_GPIO_WritePin(SR04_TRIG_GPIO_Port, SR04_TRIG_Pin, GPIO_PIN_RESET);

  return temp_distance;
}

/**
 * @brief HC_SR04超声波传感器中断处理函数，在GPIO_EXIT中断中调用
 * @param 无
 * @retval 无
 */
void HC_SR04_IRQHandler(void) {
  if (echo_state == 0) {                         // 等待上升沿
    __HAL_TIM_SET_COUNTER(&htim3, 0);            // 清零TIM3计数值
    echo_start = __HAL_TIM_GET_COUNTER(&htim3);  // 记录上升沿时刻的计数值
    echo_state = 1;                              // 进入等待下降沿状态
  } else if (echo_state == 1) {                  // 等待下降沿
    // 记录下降沿时刻的计数值
    echo_end = __HAL_TIM_GET_COUNTER(&htim3);

    // 距离 = 脉冲宽度 * 声速 / 2
    // 声速为340m/s = 0.034cm/us
    // 除以2是因为超声波需要往返一次
    temp_distance = (float)(echo_end - echo_start) * 0.034f / 2.0f;

    echo_state = 0;  // 恢复到等待上升沿状态
  }
}

    使用状态机测量HC-SR04的回波脉冲宽度。在void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)中调用HC_SR04_IRQHandler()

    二、PID

    平衡小车PID调参

    外环速度环的输出，代表为了达到目标速度，车身需要“主动倾斜”的角度。它不是一个速度值，也不是一个PWM值，而是一个角度偏移量。

    速度环给定目标速度0(静止)，经过PID后输出静止的目标角度值，传给内环直立环作为输入值，经过直立环PID输出电机驱动PWM。

    PID结构体：

    typedef struct PID_t {
    float Target, Actual, Out;
    float Kp, Ki, Kd;
    float Error0, Error1, ErrorInt;

    float OutMax, OutMin;
} PID_t;

    PID.c

    #include "PID.h"

/**
 * @brief  更新直立环PID控制器输出
 * @param  p: PID控制器结构体指针
 * @param  gyro_x: 陀螺仪X轴数据，用于微分项计算
 * @retval 无
 * 
 * 该函数实现直立环PID控制，使用陀螺仪数据作为微分反馈，
 * 通过PD控制算法计算输出值，用于保持平衡车直立状态
 */
void Vertical_PID_Update(PID_t* p, short gyro_x) {
  p->Error0 = p->Target - p->Actual;
  p->Out = p->Kp * p->Error0 - p->Kd * gyro_x;
}

/**
 * @brief  更新速度环PID控制器输出
 * @param  p: PID控制器结构体指针
 * @retval 无
 * 
 * 该函数实现速度环PID控制，包含误差计算、低通滤波、积分限幅等处理，
 * 通过PI控制算法计算输出值，用于控制平衡车的速度
 */
void Speed_PID_Update(PID_t* p) {
  const float a = 0.7f;  // 低通滤波系数

  // 计算误差
  p->Error0 = p->Target - p->Actual;

  // 滤波器
  p->Error0 = a * p->Error0 + (1 - a) * p->Error1;
  p->Error1 = p->Error0;

  p->ErrorInt += p->Error0;

  // 积分限幅
  if (p->ErrorInt > 10000) p->ErrorInt = 10000;
  if (p->ErrorInt < -10000) p->ErrorInt = -10000;

  p->Out = p->Kp * p->Error0 + p->Ki * p->ErrorInt;
}

    /**
 * @brief  更新直立环PID控制器输出
 * @param  p: PID控制器结构体指针
 * @param  gyro_x: 陀螺仪X轴数据，用于微分项计算
 * @retval 无

    需要分析确定kp，kd极性。通过测试发现，小车前后倒，改变的值为roll与x轴数据，这与陀螺仪的安装方向有关，模块上有图示。假定小车往后倒，roll为正数，gyro_x陀螺仪X轴数据也为正，电机向后运动，速度值为负，，同时设定目标角度为0，则PWM输出要求为负才能让小车平衡。

    • 确定kp极性，Error0计算得到负数，此时要让PWM输出为负，kp要大于零。
    • 确定kd极性，假设当前目标值与实际角度都为0，下一刻小车向后倒，则此时gyro_x角速度为正，想要PWM输出为负，kd需要小于零。此时PID输出为：p->Out = p->Kp * p->Error0 + p->Kd * gyro_x。为了kp，kd极性相同，可以将PID输出改为：p->Out = p->Kp * p->Error0 - p->Kd * gyro_x。当kd极性出现错误时，小车会出现平衡一小段时间，然后开始大幅度振荡

    关于为什么只用到PD，直立控制本质是“动态平衡”，系统惯性大，响应速度是关键。如果加入I项（积分），会累积历史误差，导致输出滞后，反而削弱系统的快速响应能力。

    确定kp的范围。假设PWM范围0~7200，则当Error0=10，即实际角度与目标角度相差10°时，kp=720即可输出100%占空比。所以kp范围大概是0-720。具体还需微调。当调节到能实现小车出现低频大幅震荡时即可。

    确定kd范围，kd是起到阻尼作用，实测x轴角速度为几百到一千多，故kd可以从0.1开始。直到小车出现高频的小幅震荡。

    此时将kp,kd同时乘0.6即可，此时小车不抖，但也无法平衡，此时就需要加入速度环。

    速度环

    /**
 * @brief  更新速度环PID控制器输出
 * @param  p: PID控制器结构体指针
 * @retval 无
 * 
 * 该函数实现速度环PID控制，包含误差计算、低通滤波、积分限幅等处理，
 * 通过PI控制算法计算输出值，用于控制平衡车的速度
 */
void Speed_PID_Update(PID_t* p) {
  const float a = 0.7f;  // 低通滤波系数

  // 计算误差
  p->Error0 = p->Target - p->Actual;

  // 低通滤波器
  p->Error0 = a * p->Error0 + (1 - a) * p->Error1;
  p->Error1 = p->Error0;

  p->ErrorInt += p->Error0;

  // 积分限幅
  if (p->ErrorInt > 10000) p->ErrorInt = 10000;
  if (p->ErrorInt < -10000) p->ErrorInt = -10000;

  p->Out = p->Kp * p->Error0 + p->Ki * p->ErrorInt;
}

    将PID调控放在MPU6050的中断回调函数中，MPU6050采样频率为100Hz，即10ms一次。在里面测得满速时，左右两个电机，编码器相加可达50。则kp从0.1开始调，让输出角度值变化量控制在两位数以内。kp极性确定，当kp极性错误时，会抑制小车平衡的速度，使其倒下更快。ki通过经验为kp/200。

    当小车能够在平衡的前提下出现大幅震荡，可以调节直立环的kd，抑制震荡。

    关于为什么不适用D项，速度信号（来自编码器）噪声较大，微分项（D）会放大噪声，导致系统抖动。速度环是高精度跟踪设定值，积分项是保证稳态精度关键。PD更适合位置控制或噪声敏感度低的场景。

    PID调控代码：

    uint8_t run_status = 1;  // 0:停止,1:运行PID

MPU mpu_data;
int16_t speed1, speed2, motor1_pwm, motor2_pwm;
float angle;  // 直立角度，用于直立环控制
PID_t vertical_pid = {
    .Kp = 250,  // 420
    .Ki = 0,
    .Kd = 0.46,  //0.2
    .Target = MACHINE_MEDIAN,
};  // 垂直方向速度控制PID

PID_t speed_pid = {
    .Kp = 0.7,
    .Ki = 0.0035,
    .Kd = 0,
    .Target = 0,
};  // 速度控制PID

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
    if (GPIO_Pin == MPU_INT_Pin) {  // MPU6050采样频率100Hz，每10ms触发一次中断
        if (run_status) {
            /* 内环直立环PID（10ms执行一次）*/
            MPU_DMP_ReadData(&mpu_data);
            angle = mpu_data.roll;

            /* 外环速度环*/
            speed1 = Encoder_Get(1), speed2 = Encoder_Get(2);
            speed_pid.Actual = speed1 + speed2;
            Speed_PID_Update(&speed_pid);
            vertical_pid.Target = MACHINE_MEDIAN - speed_pid.Out;

            /* 直立环 */
            vertical_pid.Actual = angle;
            Vertical_PID_Update(&vertical_pid, mpu_data.gyrox);
            motor1_pwm = vertical_pid.Out;
            motor2_pwm = vertical_pid.Out;
            Motor_SetPWM(motor1_pwm, motor2_pwm);

            // 倒下停止
            if (fabs(mpu_data.roll) > 45) {
                run_status = 0;
            }

        } else {
            Motor_SetPWM(0, 0);
        }
    }

    if (GPIO_Pin == SR04_ECHO_Pin) {  // 超声波测距
        HC_SR04_IRQHandler();
    }
}
