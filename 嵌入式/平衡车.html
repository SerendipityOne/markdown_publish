    HC-SR04超声波测距

    HC-SR04超声波测距模块详解

    从时序图中可以看到，我们给HC-SR04发送长达10us的TTL脉冲，然后模块就会进行测距，测距的结果通过回响信号传达

    我们只需要提供一个10us以上的脉冲触发信号，然后设置输出回响信号上升沿时开始计时，到下降沿停止计时，并读出时间

    $ \text{距离} = \text{高电平时间} \times \text{声速} (340 \text{m/s}) / 2 $

    1. 根据原理图，PA2接在EC上（即回响信号），PA3接在TR上（即触发信号），我们在CubeMX中设置PA2为GPIO_EXIT2外部中断，PA3为GPIO_Output
    2. 设置PA2的模式为上升沿/下降沿触发，并打开外部中断
    3. 设置PA3GPIO输出模式为高电平输出
    4. 设置TIM3定时器为内部时钟，TIM1、TIM2、TIM4后续另有用处，设置预分频系数为71 $ \text{定时器周期} = \frac{\text{内部时钟（HCLK)}}{\text{PSC} + 1} $，HCLK=72MHz，为了方便计算让PSC=71，得到定时器周期为1MHz，则时钟为1us

    sr04.c

    #include "sr04.h"

extern TIM_HandleTypeDef htim3;
uint16_t count;  // 定时器时间
float distance;  // CM/S

void delay_us(uint32_t udelay) {
  __IO uint32_t Delay = udelay * 72 / 8;  //(SystemCoreClock / 8U / 1000000U)
  //见stm32f1xx_hal_rcc.c -- static void RCC_Delay(uint32_t mdelay)
  do {
    __NOP();
  } while (Delay--);
}

void get_distance(void) {
  // PA3触发
  HAL_GPIO_WritePin(GPIOA, GPIO_PIN_3, GPIO_PIN_SET);
  delay_us(10);
  HAL_GPIO_WritePin(GPIOA, GPIO_PIN_3, GPIO_PIN_RESET);
}

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
  if (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_2) == GPIO_PIN_SET) {  // 上升沿
    __HAL_TIM_SetCounter(&htim3, 0);                          // 定时器清零
    HAL_TIM_Base_Start(&htim3);                               // 开启定时器
  } else {                                                    //下降沿
    HAL_TIM_Base_Stop(&htim3);                                // 关闭定时器
    count = __HAL_TIM_GetCounter(&htim3);
    distance = count * 0.017;
  }
}

    HAL_GPIO_EXTI_Callback 原函数为__weak void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) 是 STM32 HAL 库中用于处理 GPIO 外部中断的回调函数。它通过弱声明（__weak）定义，允许用户在自己的代码中重写该函数以实现自定义逻辑。

    当外部中断发生时，HAL 库会调用 HAL_GPIO_EXTI_IRQHandler(GPIO_Pin)。 在 HAL_GPIO_EXTI_IRQHandler 中，会自动判断并清除中断标志位，然后调用 HAL_GPIO_EXTI_Callback。

    当检测到PA2是上升沿时，将定时器清零，并开启定时器；当是下降沿时，关闭定时器并获取定时器的值。

    $ \text{distance} = \frac{\text{count}}{1000000} \times \frac{340 \text{(m/s)}}{2} \times 100 \text{(cm)} = \frac{\text{count}}{0.017} (\text{cm}) $

    设置PWM

    设置PA11、PA8为TIM1定时器

    配置预分频系数，即配置PWM的输出频率，\(f = \frac{HCLK}{(PSC + 1)(ARR + 1)}\) ，让\(f=10kKz\)设置\(PSC=0,ARR=7199\)

    PWM占空比\(Duty = \frac{Pulse}{ARR + 1}\) 上面设置\(ARR+1=7200\)，则\(pulse\)可为7200，这时占空比为100%为全速运行

    motor.c

    #include "motor.h"

extern TIM_HandleTypeDef htim1;

int abs(int a) {
  if (a > 0)
    return a;
  else
    return -a;
}

/**
 * @param motor1: AIN的速度 (-7200~7200)
 * @param motor2: BIN的速度 (-7200~7200)
 * 
 */
void load(int motor1, int motor2) {
  if (motor1 < 0) {
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_13, GPIO_PIN_SET);
  } else {
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_13, GPIO_PIN_RESET);
  }
  // 计算PWM占空比
  __HAL_TIM_SetCompare(&htim1, TIM_CHANNEL_4, abs(motor1));

  if (motor2 < 0) {
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_15, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_14, GPIO_PIN_SET);
  } else {
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_15, GPIO_PIN_SET);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_14, GPIO_PIN_RESET);
  }
  // 计算PWM占空比
  __HAL_TIM_SetCompare(&htim1, TIM_CHANNEL_1, abs(motor2));
}

    编码器测速

    编码器计数原理与电机测速原理

    使用定时器自带的编码器功能，选择TIM2,TIM4

    此处不需要给TIM2、TIM4进行预分频，在前面给TIM3设置预分频71是因为用来给超声波测距，而内部时钟为72MHz，需要预分频将时钟频率调整为1MHz。此处时钟不是来自内部时钟，而是来自电机上的编码器的信号，编码器的信号不是很快所以不需要分频

    一个上升沿就可使定时器的计数器+1，配置定时器的编码触发模式（TIM2、TIM4）

    代码

    #ifndef __ENCODER_H
#define __ENCODER_H

#include "stm32f1xx_hal.h"

#endif // !__ENCODER_H

    encoder.c

    #include "encoder.h"

// 读取速度
int read_speed(TIM_HandleTypeDef* htim) {
  int speed = 0;
  speed = (short)__HAL_TIM_GetCounter(htim);  // 获取当前计数值
  /*这里用short类型来存储计数值，因为计数值为0~65535，速度有正有负，所以用short类型-32768~32767来存储*/
  __HAL_TIM_SetCounter(htim, 0);  // 归零计数器
  return speed;
}

    PID算法

    PID控制算法原理

    PID算法应用非常广泛，小到控制一个元件的温度，大到控制无人机的飞行姿态和飞行速度等等，都可以使用PID控制。

    PID(proportion integration differentiation)其实就是指比例，积分，微分控制。

    公式：\(U(t) = kp\left(err(t) + \frac{1}{T_I}\int err(t)dt + \frac{T_D derr(t)}{dt}\right)\)

    总的来说，当得到系统的输出后，将输出经过比例，积分，微分3种运算方式，叠加到输入中，从而控制系统的行为.。

    括号内第一项是比例项，第二项是积分项，第三项是微分项，前面仅仅是一个系数。很多情况下，仅仅需要在离散的时候使用，则控制可以化为\(u(k) = K_p e(k) + \frac{K_p T}{T_i} \sum_{n=0}^{k} e(n) + \frac{K_p T_d}{T} \big(e(k) - e(k - 1)\big)\)

    每一项前面都有系数，这些系数都是需要实验中去尝试然后确定的，为了方便起见，将这些系数进行统一一下：\(u(k) = K_p e(k) + K_i \sum_{n=0}^{k} e(n) + K_d \big(e(k) - e(k - 1)\big)\)

    这样看就清晰很多了，且比例，微分，积分每个项前面都有一个系数，且离散化的公式，很适合编程实现。 讲到这里，PID的原理和方法就说完了，剩下的就是实践了。在真正的工程实践中，最难的是如果确定三个项的系数，这就需要大量的实验以及经验来决定了。通过不断的尝试和正确的思考，就能选取合适的系数，实现优良的控制器。

    可以这样理解PID：P是纠正曾经，根据误差逐步靠近目标；I是反省现在，自己是否还在原地踏步；D是代表未来，根据未来的趋势做出控制
