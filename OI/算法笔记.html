    #include <stdio.h>

// 求两个数的最大公约数，使用欧几里得算法
int gcd(int a, int b) {
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

// 函数定义：求多个数的最大公约数
int gcd_multiple_numbers(int arr[], int n) {
    int result = arr[0];
    for (int i = 1; i < n; i++) {
        result = gcd(result, arr[i]);
    }
    return result;
}

int main() {
    int numbers[] = {48, 64, 80}; // 示例数组
    int n = sizeof(numbers) / sizeof(numbers[0]);
    printf("多个数的最大公约数是: %d\n", gcd_multiple_numbers(numbers, n));
    return 0;
}

    void init_prime(bool* prime) {
  memset(prime, true, sizeof(prime));
  prime[0] = prime[1] = false;  // 0,1不是质数
  for (int i = 2; i * i <= n; i++) {
    // 如果prime[i]没有被标记为false，则它是质数
    if (prime[i]) {
      // 更新所有i的倍数为false，表示它们不是质数
      for (int j = i * i; j <= n; j += i) {
        prime[j] = false;
      }
    }
  }
}

    快速判断质数（素数）

    /**
 * @brief 快速判断一个数是否为素数
 * 
 * @param a 待判断的正整数
 * @return true 如果n是素数
 * @return false 如果n不是素数
 */
bool isPrime(int a) {
  if (a <= 1) return false;                    // 1. 处理小于等于1的数
  if (a <= 3) return true;                     // 2. 处理2和3
  if (a % 2 == 0 || a % 3 == 0) return false;  // 3. 处理能被2或3整除的数，这样做可以让我们后续的步长为6

  // 数学原理：所有大于3的素数都可以表示为6k±1的形式
  // 循环从5开始（即6*1-1），每次步长为6，这样可以同时检查6k-1和6k+1两种形式的数。
  for (int i = 5; i * i <= a; i += 6) {                // 4. 处理剩下的数，步长为6
    if (a % i == 0 || a % (i + 2) == 0) return false;  // 检查 i 和 i+2 是否是 n 的因子
  }

  return true;  // 5. 如果循环结束都没有找到因子，那么n是素数
}

    快速幂||取余 模板

    #include <stdio.h>
#define ll long long
ll a, b, p;
// 求a的b次方，快速幂算法
ll quick_pow(ll a, ll b) {
  ll ans = 1, base = a;
  while (b) {
    if (b & 1) {
      ans *= base;
      ans %= p;  // 取余
    }
    base *= base;
    base %= p;  // 取余
    b >>= 1;
  }
  return ans;
}

int main() {
  scanf("%lld%lld%lld", &a, &b, &p);
  ll ans = quick_pow(a, b);
  printf("%lld^%lld mod %lld=%lld\n", a, b, p, ans);
}

    快排模板

    #include <stdio.h>

// 交换两个整数的值
void swap(int* a, int* b) {
    int t = *a;
    *a = *b;
    *b = t;
}

// 分区函数
int partition(int arr[], int low, int high) {
    int pivot = arr[high]; // 选取最后一个元素作为基准
    int i = (low - 1); // i是较小元素的索引

    for (int j = low; j < high; j++) {
        // 如果当前元素小于或等于基准
        if (arr[j] <= pivot) {
            i++; // 增加较小元素的索引
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high]);
    return (i + 1);
}

// 快速排序函数
void quickSort(int arr[], int low, int high) {
    if (low < high) {
        // pi是分区索引，arr[pi]已经排好序
        int pi = partition(arr, low, high);

        // 分别对基准前后的子数组进行排序
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

// 打印数组的函数
void printArray(int arr[], int size) {
    for (int i = 0; i < size; i++)
        printf("%d ", arr[i]);
    printf("\n");
}

// 主函数
int main() {
    int arr[] = {10, 7, 8, 9, 1, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("原始数组: \n");
    printArray(arr, n);
    quickSort(arr, 0, n - 1);
    printf("排序后的数组: \n");
    printArray(arr, n);
    return 0;
}

    生成所有子集

    #include <iostream>
#include <vector>
using namespace std;
vector<int> v;
// 输出所有子集
void print_subsets(vector<int> v) {
  int n = v.size();  // 获取集合v的大小
  // 遍历所有可能的子集，总共有2^n个子集
  for (int i = 0; i < (1 << n); i++) {
    cout << "{ ";
    // 遍历集合中的每个元素
    for (int j = 0; j < n; j++) {
      // 检查当前子集i是否包含第j个元素
      if (i & (1 << j))
        cout << v[j] << " ";  // 如果包含，则输出该元素
    }
    cout << "}" << endl;  // 输出当前子集的结束符号
  }
}

int main() {
  int num;
  while (cin >> num) {  // 直接读取整数，忽略空格
    v.push_back(num);
  }
  print_subsets(v);
}

    原理：

    1. 子集的数量：一个包含n个元素的集合，其子集的数量为2^n。这是因为每个元素都有两种选择：要么包含在子集中，要么不包含。
    2. 二进制表示：我们可以用一个n位的二进制数来表示一个子集。例如，对于集合{2, 3, 4}，二进制数101表示子集{2, 4}，其中1表示包含该元素，0表示不包含。
    3. 遍历所有子集：通过循环for (int i = 0; i < (1 << n); i++)，我们可以遍历所有可能的子集。1 << n表示2^n，即所有子集的数量。
    4. 检查元素是否在子集中：在内部循环中，if (i & (1 << j))用于检查当前子集i是否包含第j个元素。1 << j生成一个只有第j位为1的二进制数，通过与i进行按位与操作，可以判断第j位是否为1。
    5. 输出子集：如果第j个元素在子集中，则输出该元素。最后，输出子集的结束符号}。

    但是，该方法时间复杂度为O(2n)，若要计算所有子集的和，可以使用数学公式优化

    对于集合 {a1, a2, ..., an}，所有子集的和可以通过以下公式快速计算：总和 = (a1 + a2 + ... + an) * 2^(n-1).这是因为每个元素在所有子集中出现的次数是 2^(n-1) 次。

    加法

    #include <algorithm>
#include <cstring>
#include <iostream>
using namespace std;

// 高精度加法
string add_string(string a, string b) {
  // 确保a是比b大的
  if (a.length() < b.length())
    swap(a, b);
  // 补全较短字符串的前导零
  while (b.length() < a.length())
    b = '0' + b;
  string result;
  int carry = 0;  // 进位
  // 从最低位开始逐位相加
  for (int i = a.length() - 1; i >= 0; i--) {
    int digit1 = a[i] - '0';
    int digit2 = b[i] - '0';
    int sum = digit1 + digit2 + carry;
    carry = sum / 10;
    sum %= 10;
    result.push_back(sum + '0');
  }
  // 如果最后还有进位，添加到结果中
  if (carry) {
    result.push_back(carry + '0');
  }
  reverse(result.begin(), result.end());  // 翻转
  return result;
}

int main() {
  string a, b;
  cin >> a >> b;
  cout << add_string(a, b) << endl;
}

void reverse(char* str) {

void add_str(char* a, char* b, char* res) {
  int len_a = strlen(a), len_b = strlen(b);
  int max_len = len_a > len_b ? len_a : len_b;
  int carry = 0;
  for (int i = 0; i < max_len; i++) {
    int digit_a = i < len_a ? a[len_a - 1 - i] - '0' : 0;
    int digit_b = i < len_b ? b[len_b - 1 - i] - '0' : 0;
    int sum = digit_a + digit_b + carry;
    carry = sum / 10;
    res[i] = sum % 10 + '0';
  }
  if (carry) {
    res[max_len] = carry + '0';
    res[max_len + 1] = '\0';
  } else {
    res[max_len] = '\0';
  }
  reverse(res);
}

    减法

    #include <stdio.h>
#include <string.h>

// 比较两个大整数的大小，返回1表示a > b，0表示a == b，-1表示a < b
int cmp(char* a, char* b) {
  int len_a = strlen(a), len_b = strlen(b);
  if (len_a > len_b) return 1;
  if (len_a < len_b) return -1;
  return strcmp(a, b);
}

// 高精度减法
void sub_str(char* a, char* b, char* res) {
  int len_a = strlen(a), len_b = strlen(b);
  int max_len = len_a > len_b ? len_a : len_b;
  int carry = 0;  // 借位

  // 确保a>=b
  if (cmp(a, b) < 0) {
    // 如果a < b，则交换a和b，并在结果前添加'-'
    char* t = a;
    a = b;
    b = t;

    int t_len = len_a;
    len_a = len_b;
    len_b = t_len;

    res[0] = '-';
    res++;  // 指向结果的第一个位置
  }

  // 从最低位开始相减
  for (int i = len_a - 1, j = len_b - 1, k = max_len - 1; k >= 0;
       i--, j--, k--) {
    int digit_a = i >= 0 ? a[i] - '0' : 0;
    int digit_b = j >= 0 ? b[j] - '0' : 0;
    int diff = digit_a - digit_b - carry;
    if (diff < 0) {  // 不够减
      diff += 10;
      carry = 1;
    } else {
      carry = 0;
    }
    res[k] = diff + '0';
  }

  // 去掉前导0
  int i = 0;
  while (i < max_len - 1 && res[i] == '0') {
    i++;
  }
  if (i > 0) {
    // 将去掉前导零后的字符移动到 res 的开头
    // res: 目标内存区域的起始地址。
    // res+i: 源内存区域的起始地址，即去掉前导零后的第一个字符的位置。
    // max_len-i+1:要复制的字节数，即去掉前导零后剩余的字符数加一（包括字符串结束符 \0）。 
    memmove(res, res + i, max_len - i + 1);
  }
}

int main() {
  scanf("%s%s", a, b);
  sub_str(a, b, res);
  printf("%s\n", res);
}

    #include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LEN 2005

// 反转字符串
void reverseString(char* str) {
  int len = strlen(str);
  for (int i = 0; i < len / 2; i++) {
    char temp = str[i];
    str[i] = str[len - 1 - i];
    str[len - 1 - i] = temp;
  }
}

// 高精度乘法
char* multiplyStrings(const char* num1, const char* num2) {
  // 有一个是0
  if (strcmp(num1, "0") == 0 || strcmp(num2, "0") == 0) {
    char* result = (char*)malloc(2 * sizeof(char));
    strcpy(result, "0");
    return result;
  }

  int len1 = strlen(num1);
  int len2 = strlen(num2);
  int resultLen = len1 + len2;
  // 用于存储中间计算结果
  int* result = (int*)calloc(resultLen, sizeof(int));

  // 反转数字字符串以便从低位开始计算
  char* n1 = strdup(num1);  // 复制字符串以免修改原字符串
  char* n2 = strdup(num2);
  reverseString(n1);
  reverseString(n2);

  // 逐位相乘
  for (int i = 0; i < len1; i++) {
    for (int j = 0; j < len2; j++) {
      // 表示该乘积应该放在哪个位置（类似于手工乘法的对齐方式）
      result[i + j] += (n1[i] - '0') * (n2[j] - '0');
    }
  }

  // 处理进位
  // 例如，如果 result[2] = 12
  // result[3] += 12 / 10 = 1（进位）
  // result[2] = 12 % 10 = 2（保留个位数）
  for (int i = 0; i < resultLen - 1; i++) {
    result[i + 1] += result[i] / 10;
    result[i] %= 10;
  }

  // 转换为字符串
  char* product = (char*)malloc((resultLen + 1) * sizeof(char));
  int index = 0;
  int leadingZero = 1;  // 用于跳过前导零
  for (int i = resultLen - 1; i >= 0; i--) {
    if (leadingZero && result[i] == 0) {
      continue;
    }
    leadingZero = 0;
    product[index++] = result[i] + '0';
  }
  product[index] = '\0';

  free(n1);
  free(n2);
  free(result);

  return product;
}

int main() {
  char num1[MAX_LEN];
  char num2[MAX_LEN];
  scanf("%s%s", num1, num2);

  char* product = multiplyStrings(num1, num2);
  printf("%s\n", product);
  free(product);

  return 0;
}

    除法

    #include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LEN 5005

// 移除字符串前导零
void removeLeadingZeros(char* str) {
  int len = strlen(str);
  int nonZeroIndex = 0;

  // 找到第一个非零字符的位置
  while (nonZeroIndex < len && str[nonZeroIndex] == '0') {
    nonZeroIndex++;
  }

  // 如果全是零，保留一个零
  if (nonZeroIndex == len) {
    str[0] = '0';
    str[1] = '\0';
    return;
  }

  // 移动剩余字符到开头
  memmove(str, str + nonZeroIndex, len - nonZeroIndex + 1);
}

// 比较两个大数的大小，返回1表示a>b，0表示a==b，-1表示a<b
int compare(const char* a, const char* b) {
  int lenA = strlen(a);
  int lenB = strlen(b);

  if (lenA > lenB) return 1;
  if (lenA < lenB) return -1;

  return strcmp(a, b);
}

// 大数减法 (a - b)，假设a >= b
void subtract(char* a, const char* b) {
  int lenA = strlen(a);
  int lenB = strlen(b);

  // 从低位开始减
  int carry = 0;
  for (int i = 0; i < lenA; i++) {
    int digitA = a[lenA - 1 - i] - '0';
    int digitB = (i < lenB) ? (b[lenB - 1 - i] - '0') : 0;

    digitA -= carry;
    if (digitA < digitB) {
      digitA += 10;
      carry = 1;
    } else {
      carry = 0;
    }

    a[lenA - 1 - i] = (digitA - digitB) + '0';
  }

  removeLeadingZeros(a);
}

// 高精度除法，返回商和余数
void divideStrings(const char* dividend,
                   const char* divisor,
                   char* quotient,
                   char* remainder) {
  // 处理除数为0的情况
  if (strcmp(divisor, "0") == 0) {
    printf("Error: Division by zero\n");
    exit(1);
  }

  // 处理被除数为0的情况
  if (strcmp(dividend, "0") == 0) {
    strcpy(quotient, "0");
    strcpy(remainder, "0");
    return;
  }

  // 比较大小
  int cmp = compare(dividend, divisor);
  if (cmp < 0) {
    // 被除数小于除数
    strcpy(quotient, "0");
    strcpy(remainder, dividend);
    return;
  } else if (cmp == 0) {
    // 被除数等于除数
    strcpy(quotient, "1");
    strcpy(remainder, "0");
    return;
  }

  int lenDividend = strlen(dividend);
  int lenDivisor = strlen(divisor);

  // 初始化余数为空
  char currentRemainder[MAX_LEN + 1] = {0};

  // 初始化商
  int quotientIndex = 0;

  // 逐位处理被除数
  for (int i = 0; i < lenDividend; i++) {
    // 将当前位添加到余数
    int len = strlen(currentRemainder);
    currentRemainder[len] = dividend[i];
    currentRemainder[len + 1] = '\0';

    removeLeadingZeros(currentRemainder);

    // 如果当前余数小于除数，商位为0
    if (compare(currentRemainder, divisor) < 0) {
      quotient[quotientIndex++] = '0';
      continue;
    }

    // 计算当前位的商
    int q = 0;
    char tempDivisor[MAX_LEN + 1];
    strcpy(tempDivisor, divisor);

    while (compare(currentRemainder, tempDivisor) >= 0) {
      subtract(currentRemainder, tempDivisor);
      q++;
    }

    quotient[quotientIndex++] = q + '0';
  }

  quotient[quotientIndex] = '\0';
  removeLeadingZeros(quotient);
  strcpy(remainder, currentRemainder);

  // 如果余数为空，设为0
  if (strlen(remainder) == 0) {
    strcpy(remainder, "0");
  }
}

int main() {
  char dividend[MAX_LEN];   // 被除数
  char divisor[MAX_LEN];    // 除数
  char quotient[MAX_LEN];   // 商
  char remainder[MAX_LEN];  // 余数

  scanf("%s%s", dividend, divisor);

  // 验证输入
  for (int i = 0; dividend[i]; i++) {
    if (dividend[i] < '0' || dividend[i] > '9') {
      printf("错误: 被除数包含非数字字符\n");
      return 1;
    }
  }
  for (int i = 0; divisor[i]; i++) {
    if (divisor[i] < '0' || divisor[i] > '9') {
      printf("错误: 除数包含非数字字符\n");
      return 1;
    }
  }

  divideStrings(dividend, divisor, quotient, remainder);

  printf("商: %s\n", quotient);
  printf("余数: %s\n", remainder);

  return 0;
}

    求阶乘之和

    #include <stdio.h>
#include <string.h>

#define MAX_DIGITS 5000  // 表示大整数最多可以有多少位

// 高精度乘法：将一个大整数a与一个小整数b相乘，结果存储在a中
void multiply(int a[], int* len, int b) {
  int carry = 0;
  for (int i = 0; i < *len; i++) {
    int product = a[i] * b + carry;  // 计算当前位的乘积
    a[i] = product % 10;             // 取余数作为当前位的值
    carry = product / 10;            // 取整作为进位值
  }
  while (carry) {  // 处理最高位的进位值
    a[*len] = carry % 10;
    carry /= 10;
    (*len)++;
  }
}

// 高精度加法：将两个大整数a和b相加，结果存储在a中
void add(int a[], int* len_a, int b[], int len_b) {
  int carry = 0;
  int max_len = (*len_a > len_b) ? *len_a : len_b;
  for (int i = 0; i < max_len; i++) {
    int sum = a[i] + b[i] + carry;
    a[i] = sum % 10;
    carry = sum / 10;
  }
  if (carry) {
    a[max_len] = carry;
    (*len_a) = max_len + 1;
  } else {
    (*len_a) = max_len;
  }
}

// 计算阶乘之和
void factorial_sum(int n, int result[], int* len_result) {
  int factorial[MAX_DIGITS] = {1};
  int len_fact = 1;

  for (int i = 1; i <= n; i++) {
    multiply(factorial, &len_fact, i);             // 计算i!
    add(result, len_result, factorial, len_fact);  // 加到阶乘之和中
  }
}

int main() {
  int n;
  scanf("%d", &n);

  int result[MAX_DIGITS] = {0};  // 存储阶乘之和，反转来存
  int len_result = 1;

  factorial_sum(n, result, &len_result);

  for (int i = len_result - 1; i >= 0; i--) {
    printf("%d", result[i]);
  }
  printf("\n");

  return 0;
}
